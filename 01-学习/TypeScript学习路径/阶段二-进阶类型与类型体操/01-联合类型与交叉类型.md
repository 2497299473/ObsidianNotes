## 1. 联合类型（Union Types）

### 1.1 基本用法

```typescript
// 一个值可以是多种类型之一
type ID = string | number;

function findUser(id: ID) {
  if (typeof id === "string") {
    // id 在此分支收窄为 string
    return db.queryByText(id);
  }
  // id 在此分支收窄为 number
  return db.queryById(id);
}
```

### 1.2 字面量联合类型

```typescript
type Direction = "up" | "down" | "left" | "right";
type HttpStatus = 200 | 404 | 500;

function move(dir: Direction) {
  // dir 只能是四个字面量之一
}
```

### 1.3 联合类型的成员

联合类型 `A | B | C` 的成员是 A、B、C。对联合类型调用方法时，**只能调用所有成员共有的方法**：

```typescript
interface Bird { fly(): void; layEggs(): void; }
interface Fish { swim(): void; layEggs(): void; }

type Pet = Bird | Fish;

function care(pet: Pet) {
  pet.layEggs(); // ✅ 两者都有
  // pet.fly();  // ❌ Fish 没有 fly
  // pet.swim(); // ❌ Bird 没有 swim
}
```

## 2. 交叉类型（Intersection Types）

### 2.1 基本用法

```typescript
interface WithId { id: number; }
interface WithTimestamp { createdAt: Date; }

type Record = WithId & WithTimestamp;
// Record 同时拥有 id 和 createdAt
```

### 2.2 与 interface extends 对比

```typescript
// 用交叉类型
type A = { a: string } & { b: number };
// 等价于
interface B extends { a: string } { b: number } // ❌ 语法错误
// 正确写法
interface Base { a: string; }
interface B extends Base { b: number; }
```

### 2.3 同名属性的交叉行为

```typescript
type T1 = { name: string };
type T2 = { name: "Alice" };

type Merged = T1 & T2; // { name: "Alice" } —— 字面量收窄
type Conflict = { age: string } & { age: number };
// age 的类型变为 never —— 实际中应避免
```

## 3. 可辨识联合（Discriminated Unions）

这是 TypeScript 中最重要的模式之一，用于建模状态机。

### 3.1 基本模式

```typescript
// 每个 variant 有一个共同的字面量类型字段（tag/discriminant）
type NetworkState =
  | { status: "idle" }
  | { status: "loading" }
  | { status: "success"; data: string }
  | { status: "error"; error: Error };

function handleState(state: NetworkState) {
  switch (state.status) {
    case "idle":
      return "等待中";
    case "loading":
      return "加载中";
    case "success":
      return state.data; // ✅ TS 知道此时 data 存在
    case "error":
      return state.error.message; // ✅ TS 知道此时 error 存在
  }
}
```

### 3.2 穷尽性检查

```typescript
// 利用 never 类型确保 switch 覆盖了所有情况
function assertNever(x: never): never {
  throw new Error(`Unexpected: ${x}`);
}

function handleStateExhaustive(state: NetworkState) {
  switch (state.status) {
    case "idle": return null;
    case "loading": return null;
    case "success": return state.data;
    case "error": return state.error.message;
    default:
      return assertNever(state); // 如果遗漏一个 case，这里会报错
  }
}
```

## 4. 类型收窄（Type Narrowing）

### 4.1 typeof 收窄

```typescript
function format(value: string | number) {
  if (typeof value === "string") {
    return value.toUpperCase(); // value: string
  }
  return value.toFixed(2); // value: number
}
```

### 4.2 in 操作符收窄

```typescript
interface Cat { meow(): void; }
interface Dog { bark(): void; }

function speak(animal: Cat | Dog) {
  if ("meow" in animal) {
    animal.meow(); // animal: Cat
  } else {
    animal.bark(); // animal: Dog
  }
}
```

### 4.3 instanceof 收窄

```typescript
function processDate(value: string | Date) {
  if (value instanceof Date) {
    return value.toISOString(); // value: Date
  }
  return value.toUpperCase(); // value: string
}
```
