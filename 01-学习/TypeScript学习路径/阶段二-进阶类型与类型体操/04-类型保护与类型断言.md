## 1. 类型守卫（Type Guards）

### 1.1 typeof 守卫

```typescript
function format(value: string | number | boolean): string {
  if (typeof value === "string") {
    return value.toUpperCase(); // value 收窄为 string
  } else if (typeof value === "number") {
    return value.toFixed(2); // value 收窄为 number
  } else {
    return value ? "yes" : "no"; // value 收窄为 boolean
  }
}
```

> **typeof 只支持**：`string`、`number`、`boolean`、`bigint`、`symbol`、`undefined`、`function`、`object`。
> 不能用来判断自定义类型。

### 1.2 instanceof 守卫

```typescript
class ApiError extends Error {
  code: number;
  constructor(msg: string, code: number) {
    super(msg);
    this.code = code;
  }
}

function handleError(err: Error | ApiError) {
  if (err instanceof ApiError) {
    console.log(err.code); // ✅ 收窄为 ApiError
  } else {
    console.log(err.message);
  }
}
```

### 1.3 in 守卫

```typescript
interface Cat { meow(): void; kind: "cat"; }
interface Dog { bark(): void; kind: "dog"; }

function speak(animal: Cat | Dog) {
  if ("meow" in animal) {
    animal.meow(); // ✅ 收窄为 Cat
  } else {
    animal.bark(); // ✅ 收窄为 Dog
  }
}
```

### 1.4 自定义类型守卫（Type Predicates）

```typescript
interface User {
  name: string;
  email: string;
}

// is 关键字定义类型守卫
function isUser(obj: unknown): obj is User {
  return (
    typeof obj === "object" &&
    obj !== null &&
    "name" in obj &&
    "email" in obj &&
    typeof (obj as User).name === "string" &&
    typeof (obj as User).email === "string"
  );
}

function process(data: unknown) {
  if (isUser(data)) {
    console.log(data.name); // ✅ 此处 data 为 User
  }
}
```

### 1.5 可辨识联合的穷尽检查

```typescript
type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "rect"; width: number; height: number }
  | { kind: "triangle"; base: number; height: number };

function area(shape: Shape): number {
  switch (shape.kind) {
    case "circle":
      return Math.PI * shape.radius ** 2;
    case "rect":
      return shape.width * shape.height;
    case "triangle":
      return (shape.base * shape.height) / 2;
    default:
      const _exhaustive: never = shape;
      return _exhaustive; // 如果遗漏某个 kind，此处报错
  }
}
```

## 2. 类型断言（Type Assertions）

### 2.1 as 语法

```typescript
const el = document.getElementById("app") as HTMLDivElement;
const data = JSON.parse(raw) as { name: string };
```

### 2.2 as const 断言

```typescript
// 普通字面量 — 推断为宽泛类型
const arr = [1, 2, 3];       // number[]
const obj = { x: 1, y: 2 }; // { x: number; y: number }

// as const — 推断为最精确的字面量类型 + readonly
const arrC = [1, 2, 3] as const;       // readonly [1, 2, 3]
const objC = { x: 1, y: 2 } as const; // { readonly x: 1; readonly y: 2 }
```

### 2.3 双重断言

```typescript
// 当两个类型没有交集时，单次 as 会报错
// const s: string = 42 as string; // ❌ 编译报错

// 双重断言可以绕过（不推荐，仅用于特殊情况）
const s: string = 42 as unknown as string;
```

> **⚠️ 警告**：双重断言是"我知道我在做什么"的信号。90% 的情况下说明设计有问题。

### 2.4 非空断言（!）

```typescript
function getFirst(arr: number[]): number {
  return arr[0]!; // 你确定数组不为空
}

const el = document.getElementById("btn")!;
el.addEventListener("click", handler); // 你确定元素存在
```

## 3. satisfies 操作符（TS 4.9+）

### 3.1 as vs satisfies

```typescript
// as — 断言，会丢失类型信息
const config1 = {
  width: 100,
  color: "red",
} as Record<string, string | number>;
config1.width; // string | number — 丢失了精确类型

// satisfies — 验证，保留类型信息
const config2 = {
  width: 100,
  color: "red",
} satisfies Record<string, string | number>;
config2.width; // number — 保留了精确类型！
```

### 3.2 satisfies 的心智模型

`expr satisfies T` 做两件事：

1. **校验**：`expr` 必须可赋给 `T`，否则编译报错（包括对象字面量的多余属性检查）；
2. **不改类型**：表达式的结果类型仍是**推断出的那个类型**，不会变成 `T`。

这正是与 `: T` 标注的本质区别——标注会把变量类型**固定**为 `T`，推断出的更精确信息被抹掉；`satisfies` 只做校验，精确信息保留。

```typescript
type Theme = {
  primary: string;
  secondary: string;
  spacing: number;
};

// satisfies：校验结构，但 theme 的类型仍是推断结果，不是 Theme
const theme = {
  primary: "#ff0000",
  secondary: "#00ff00",
  spacing: 8,
} satisfies Theme;

// ⚠️ 常见误解：theme.primary 并不是 "#ff0000"，仍是 string
// （对象字面量属性默认拓宽为 string/number，satisfies 不阻止拓宽）

// 与 : T 标注的差别在"更精确"的场景才显现：
type Env = { mode: "dev" | "prod" };
const e1: Env = { mode: "dev" };          // e1.mode: "dev" | "prod"（被固定为 Env）
const e2 = { mode: "dev" } satisfies Env; // e2.mode: "dev"（推断的字面量保留）

// Record 场景：标注后所有属性访问都退化成索引签名类型
const r1: Record<string, string | number> = { width: 100 };
// r1.width 是 string | number，不能直接当 number 用
const r2 = { width: 100 } satisfies Record<string, string | number>;
const w2: number = r2.width; // ✅ 仍是 number

// 想让属性保留字面量类型，需要 as const 配合：
const themeConst = {
  primary: "#ff0000",
  secondary: "#00ff00",
  spacing: 8,
} as const satisfies Theme;
// 此时 themeConst.primary 才是 "#ff0000"
```

## 4. 类型安全的 JSON 解析

```typescript
interface UserSchema {
  id: number;
  name: string;
  roles: string[];
}

function parseUser(json: string): UserSchema {
  const data: unknown = JSON.parse(json);

  if (
    typeof data !== "object" ||
    data === null ||
    typeof (data as any).id !== "number" ||
    typeof (data as any).name !== "string" ||
    !Array.isArray((data as any).roles)
  ) {
    throw new Error("Invalid user data");
  }

  return data as UserSchema; // 经过运行时检查后安全断言
}
```

> **最佳实践**：用 `zod` 或 `valibot` 等库替代手动类型守卫做 JSON 解析。
