## 📋 改造任务

### 1. 引入不可变性

```typescript
// v1: 可变数据——直接修改
function completeTask(id: number): void {
  const task = tasks.find((t) => t.id === id);
  if (task) task.completed = true; // 直接修改
}

// v2: 不可变——返回新数组
function completeTask(tasks: readonly Task[], id: number): readonly Task[] {
  return tasks.map((t) => (t.id === id ? { ...t, completed: true } : t));
}
```

### 2. 纯函数——所有状态作为参数传入

```typescript
interface TaskState {
  tasks: readonly Task[];
  nextId: number;
}

function addTask(state: TaskState, title: string, priority: Priority = "normal"): TaskState {
  const task: Task = {
    id: state.nextId,
    title,
    completed: false,
    priority,
    createdAt: new Date(),
  };
  return {
    tasks: [...state.tasks, task],
    nextId: state.nextId + 1,
  };
}

function removeTask(state: TaskState, id: number): TaskState {
  return {
    ...state,
    tasks: state.tasks.filter((t) => t.id !== id),
  };
}
```

### 3. 高阶函数——类型安全的谓词

```typescript
// 谓词类型别名
type TaskPredicate = (task: Task) => boolean;

// 通用过滤函数
function filterTasks(tasks: readonly Task[], predicate: TaskPredicate): readonly Task[] {
  return tasks.filter(predicate);
}

// 预置谓词工厂
const byPriority = (p: Priority): TaskPredicate => (t) => t.priority === p;
const byCompleted = (completed: boolean): TaskPredicate => (t) => t.completed === completed;

// 谓词组合
const and = (...predicates: TaskPredicate[]): TaskPredicate =>
  (t) => predicates.every((p) => p(t));

const or = (...predicates: TaskPredicate[]): TaskPredicate =>
  (t) => predicates.some((p) => p(t));

// 使用
const highPriorityCompleted = filterTasks(
  state.tasks,
  and(byPriority("high"), byCompleted(true))
);
```

### 4. 可辨识联合——事件驱动

```typescript
type TaskEvent =
  | { type: "add"; title: string; priority: Priority }
  | { type: "complete"; id: number }
  | { type: "remove"; id: number }
  | { type: "update"; id: number; title: string };

// Reducer 风格的状态更新
function reduce(state: TaskState, event: TaskEvent): TaskState {
  switch (event.type) {
    case "add":
      return addTask(state, event.title, event.priority);
    case "complete":
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === event.id ? { ...t, completed: true } : t
        ),
      };
    case "remove":
      return removeTask(state, event.id);
    case "update":
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === event.id ? { ...t, title: event.title } : t
        ),
      };
    default:
      const _exhaustive: never = event;
      return _exhaustive;
  }
}
```
