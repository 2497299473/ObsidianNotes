## 📋 需求：任务管理器

管理一组待办任务，支持增删改查、按状态筛选、统计。

### 原始 JS 代码

```javascript
const tasks = [];
let nextId = 1;

function addTask(title, priority = "normal") {
  const task = {
    id: nextId++,
    title,
    completed: false,
    priority,
    createdAt: new Date(),
  };
  tasks.push(task);
  return task;
}

function completeTask(id) {
  const task = tasks.find((t) => t.id === id);
  if (task) task.completed = true;
}

function removeTask(id) {
  const idx = tasks.findIndex((t) => t.id === id);
  if (idx !== -1) tasks.splice(idx, 1);
}

function listTasks(filter = "all") {
  if (filter === "completed") return tasks.filter((t) => t.completed);
  if (filter === "pending") return tasks.filter((t) => !t.completed);
  return tasks;
}

function getStats() {
  return {
    total: tasks.length,
    completed: tasks.filter((t) => t.completed).length,
    pending: tasks.filter((t) => !t.completed).length,
  };
}

// 使用
addTask("学习 TypeScript", "high");
addTask("写毕业项目", "high");
completeTask(1);
console.log(getStats());
```

## ✅ 验收标准

| 检查项 | 方式 |
|--------|------|
| 所有变量/参数/返回值有类型标注 | `tsc --strict` 零错误 |
| 用 `interface` 定义 Task | 类型定义 |
| 用字面量联合定义 `Priority` 和 `Filter` | 非 string |
| 数组操作类型安全 | `find`/`filter`/`splice` 返回类型正确 |
| 可选参数正确处理 | `priority` 默认值、`filter` 默认值 |

```typescript

type Priority = "low" | "normal" | "high";

interface Task {
  id: number;

  completed: boolean;
  priority: Priority;

type Filter = "all" | "completed" | "pending";

interface Stats {

  pending: number;

let nextId = 1;

}
```
