## 🎯 v4 目标

为任务管理器编写一套**类型工具库**，用类型体操提升类型层面的表达力。这是"能不能做高级类型设计"的判定。

## 📋 改造任务

### 1. 实体相关工具类型

```typescript
// 创建实体时的输入——排除 id、createdAt 等系统字段
type CreateInput<T> = Omit<T, "id" | "createdAt">;

// 更新实体时的输入——排除系统字段，其余可选
type UpdateInput<T> = Partial<CreateInput<T>>;

// 实体快照——所有字段只读
type Snapshot<T> = Readonly<{
  [K in keyof T]: T[K] extends object ? Readonly<T[K]> : T[K];
}>;

// 应用到 Task
type TaskCreateInput = CreateInput<Task>;    // { title: string; completed: boolean; priority: Priority }
type TaskUpdateInput = UpdateInput<Task>;    // { title?: string; completed?: boolean; priority?: Priority }
type TaskSnapshot = Snapshot<Task>;          // 所有字段 readonly
```

### 2. 可选/必选字段提取

```typescript
// 提取可选字段
type OptionalKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? K : never;
}[keyof T];

// 提取必选字段
type RequiredKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? never : K;
}[keyof T];

// 按值类型过滤属性
type PickByType<T, V> = {
  [K in keyof T as T[K] extends V ? K : never]: T[K];
};

// 应用
type TaskStringFields = PickByType<Task, string>;
// 结果：{ title: string; priority: Priority }
// ⚠️ 注意 priority 也会被选中——字面量联合 "low" | "normal" | "high" 同样 extends string。
// 想只选"纯 string 字段"，把条件改成双向判断：T[K] extends V ? (V extends T[K] ? K : never) : never
```

### 3. 事件类型自动生成

```typescript
// 自动生成 CRUD 事件类型
type CrudEvent<T extends Identifiable> =
  | { type: "create"; data: CreateInput<T> }
  | { type: "update"; id: number; patch: UpdateInput<T> }
  | { type: "delete"; id: number }
  | { type: "read"; id: number };

type TaskEvent = CrudEvent<Task>;

// 事件分发器
function dispatch<T extends Identifiable>(
  state: EntityState<T>,
  event: CrudEvent<T>
): EntityState<T> {
  switch (event.type) {
    case "create":
      return createEntity(state, event.data);
    case "update":
      return updateEntity(state, event.id, event.patch);
    case "delete":
      return removeEntity(state, event.id);
    case "read":
      return state; // read 不改变状态
    default:
      const _exhaustive: never = event;
      return _exhaustive;
  }
}
```

### 4. 模板字面量类型——事件监听器

```typescript
// 事件名 → 监听器类型映射
type Listeners<T extends Identifiable> = {
  [E in CrudEvent<T>["type"] as `on${Capitalize<E>}`]: (
    event: Extract<CrudEvent<T>, { type: E }>
  ) => void;
};

type TaskListeners = Listeners<Task>;
// {
//   onCreate: (event: { type: "create"; data: CreateInput<Task> }) => void;
//   onUpdate: (event: { type: "update"; id: number; patch: UpdateInput<Task> }) => void;
//   onDelete: (event: { type: "delete"; id: number }) => void;
//   onRead:   (event: { type: "read"; id: number }) => void;
// }

// 使用——类型完全安全
const listeners: TaskListeners = {
  onCreate: (e) => { /* e.data: CreateInput<Task> */ },
  onUpdate: (e) => { /* e.id: number; e.patch: UpdateInput<Task> */ },
  onDelete: (e) => { /* e.id: number */ },
  onRead:   (e) => { /* e.id: number */ },
};
```

### 5. 递归工具类型

```typescript
// DeepPartial —— 递归可选
type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K];
};

// DeepReadonly —— 递归只读
type DeepReadonly<T> = {
  readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K];
};

// DeepRequired —— 递归必选
type DeepRequired<T> = {
  [K in keyof T]-?: T[K] extends object ? DeepRequired<T[K]> : T[K];
};
```

### 6. 路径类型——嵌套属性访问

```typescript
// 获取嵌套对象路径
type Paths<T> = T extends object
  ? {
      [K in keyof T & string]: K | `${K}.${Paths<T[K]>}`;
    }[keyof T & string]
  : never;

// 根据路径字符串获取嵌套类型
type Get<T, P extends string> = P extends `${infer K}.${infer Rest}`
  ? K extends keyof T
    ? Get<T[K], Rest>
    : never
  : P extends keyof T
    ? T[P]
    : never;
```
