### 4. utils/type-tools.ts — 工具类型

```typescript
export type CreateInput<T> = Omit<T, "id" | "createdAt">;
export type UpdateInput<T> = Partial<CreateInput<T>>;
export type Snapshot<T> = Readonly<{ [K in keyof T]: T[K] }>;

export type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K];
};

export type DeepReadonly<T> = {
  readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K];
};
```

### 5. core/entity-manager.ts — 泛型实体管理器

```typescript
import type { Identifiable } from "../types/base";
import type { CreateInput, UpdateInput } from "../utils/type-tools";

export interface EntityState<T extends Identifiable> {
  items: readonly T[];
  nextId: number;
}

export function createInitialState<T extends Identifiable>(): EntityState<T> {
  return { items: [], nextId: 1 };
}

export function createEntity<T extends Identifiable>(
  state: EntityState<T>,
  data: CreateInput<T>
): EntityState<T> {
  // createdAt 与 id 一样是系统字段，由这里统一填充（调用方不要传）；
  // 经 unknown 中转规避 TS2352（交叉类型与泛型 T 不充分重叠）
  const entity = { ...data, id: state.nextId, createdAt: new Date() } as unknown as T;
  return {
    items: [...state.items, entity],
    nextId: state.nextId + 1,
  };
}

export function updateEntity<T extends Identifiable>(
  state: EntityState<T>,
  id: number,
  patch: UpdateInput<T>
): EntityState<T> {
  return {
    ...state,
    items: state.items.map((item) =>
      item.id === id ? ({ ...item, ...patch } as T) : item
    ),
  };
}

export function removeEntity<T extends Identifiable>(
  state: EntityState<T>,
  id: number
): EntityState<T> {
  return {
    ...state,
    items: state.items.filter((item) => item.id !== id),
  };
}

export function findById<T extends Identifiable>(
  state: EntityState<T>,
  id: number
): T | undefined {
  return state.items.find((item) => item.id === id);
}
```

### 6. services/task-service.ts — Task 业务层

```typescript
import type { Task, Priority, Filter, Stats } from "../types";
import type { EntityState } from "../core/entity-manager";
import {
  createInitialState,
  createEntity,
  updateEntity,
  removeEntity,
} from "../core/entity-manager";

export function createTaskState(): EntityState<Task> {
  return createInitialState<Task>();
}

export function addTask(
  state: EntityState<Task>,
  title: string,
  priority: Priority = "normal"
): EntityState<Task> {
  // createdAt 由 createEntity 统一填充，这里传了反而报 TS2353（多余属性）
  return createEntity(state, {
    title,
    priority,
    completed: false,
  });
}

export function completeTask(state: EntityState<Task>, id: number): EntityState<Task> {
  return updateEntity(state, id, { completed: true });
}

export function removeTask(state: EntityState<Task>, id: number): EntityState<Task> {
  return removeEntity(state, id);
}

export function listTasks(state: EntityState<Task>, filter: Filter = "all"): readonly Task[] {
  switch (filter) {
    case "completed": return state.items.filter((t) => t.completed);
    case "pending": return state.items.filter((t) => !t.completed);
    default: return state.items;
  }
}

export function getStats(state: EntityState<Task>): Stats {
  return {
    total: state.items.length,
    completed: state.items.filter((t) => t.completed).length,
    pending: state.items.filter((t) => !t.completed).length,
  };
}
```

### 8. tsconfig 路径别名

> ⚠️ **重要**：`tsc` 只做类型检查，**不会在编译输出中转换路径别名**。配置 `paths` 后 `tsc --noEmit` 能通过类型检查，但直接运行 `node dist/index.js` 会报 `Cannot find module '@types/...'`。运行时需打包工具或 `tsc-alias` 处理。

```jsonc
{
  "compilerOptions": {
    "target": "ES2020",
    "module": "ESNext",
    "moduleResolution": "bundler",
    "strict": true,
    "esModuleInterop": true,
    "declaration": true,
    "outDir": "./dist",
    "rootDir": "./src",
    "baseUrl": ".",
    "paths": {
      "@types/*": ["./src/types/*"],
      "@utils/*": ["./src/utils/*"],
      "@core/*": ["./src/core/*"]
    }
  },
  "include": ["src/**/*"]
}
```

### 9. import type 的正确使用

```typescript
// ✅ 类型导入用 import type——编译后完全擦除
import type { Task, Priority } from "@types/index";

// ✅ 混合导入（TS 4.5+）
import { type Task, createInitialState } from "./exports";

// ✅ 类型与逻辑分离：types/ 只放类型，core/ 只放逻辑
// 避免循环依赖：类型层不导入逻辑层
```
