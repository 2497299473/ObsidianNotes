## 🎯 v3 目标

用泛型把 v2 的任务管理器抽象为**通用的实体管理器**。这是"能不能复用设计"的判定。

## 📋 改造任务

### 1. 泛型 Entity 约束

```typescript
// 所有实体的基础约束——必须有 id
interface Identifiable {
  id: number;
}

// 泛型状态
interface EntityState<T extends Identifiable> {
  items: readonly T[];
  nextId: number;
}

function createInitialState<T extends Identifiable>(): EntityState<T> {
  return { items: [], nextId: 1 };
}
```

### 2. 泛型 CRUD 操作

```typescript
// 泛型 create——排除 id（由系统生成）
function createEntity<T extends Identifiable>(
  state: EntityState<T>,
  data: Omit<T, "id" | "createdAt">
): EntityState<T> {
  // ⚠️ 不能直接 `as T`：{ ...data, id, createdAt } 与泛型 T 不充分重叠（TS2352），
  // T 可能实例化为带更多必选字段的子类型；需经 unknown 中转
  const entity = { ...data, id: state.nextId, createdAt: new Date() } as unknown as T;
  return {
    items: [...state.items, entity],
    nextId: state.nextId + 1,
  };
}

// 泛型 update——部分更新，排除 id
function updateEntity<T extends Identifiable>(
  state: EntityState<T>,
  id: number,
  patch: Partial<Omit<T, "id" | "createdAt">>
): EntityState<T> {
  return {
    ...state,
    items: state.items.map((item) =>
      item.id === id ? { ...item, ...patch } as T : item
    ),
  };
}

// 泛型 remove
function removeEntity<T extends Identifiable>(
  state: EntityState<T>,
  id: number
): EntityState<T> {
  return {
    ...state,
    items: state.items.filter((item) => item.id !== id),
  };
}
```

### 3. 泛型查询——类型安全的属性查找

```typescript
function findById<T extends Identifiable>(
  state: EntityState<T>,
  id: number
): T | undefined {
  return state.items.find((item) => item.id === id);
}

// keyof T 保证 key 和 value 类型对齐
function findByProperty<T extends Identifiable, K extends keyof T>(
  state: EntityState<T>,
  key: K,
  value: T[K]
): readonly T[] {
  return state.items.filter((item) => item[key] === value);
}

// 使用——key 和 value 类型完全安全
const highPriorityTasks = findByProperty(taskState, "priority", "high");
// ✅ key 只能是 Task 的属性名
// ✅ value 必须匹配该属性的类型
// findByProperty(taskState, "nonexistent", 1); // ❌ 编译报错
```

### 4. Task 接入泛型管理器

```typescript
interface Task extends Identifiable {
  title: string;
  completed: boolean;
  priority: Priority;
  createdAt: Date;
}

let taskState = createInitialState<Task>();
taskState = createEntity(taskState, {
  title: "学习 TypeScript",
  completed: false,
  priority: "high",
});
taskState = updateEntity(taskState, 1, { completed: true });
```

### 5. 扩展——多实体类型

```typescript
interface BugReport extends Identifiable {
  description: string;
  severity: "low" | "medium" | "high";
  resolved: boolean;
}

let bugState = createInitialState<BugReport>();
bugState = createEntity(bugState, {
  description: "页面崩溃",
  severity: "high",
  resolved: false,
});
// 同一套泛型代码，管理不同实体类型
```

## ✅ 验收标准

| 检查项 | 方式 |
|--------|------|
| `createEntity<T>` 泛型，T 用 `extends Identifiable` 约束 | `tsc --strict` |
| `create` 参数用 `Omit<T, "id" | "createdAt">` | 排除系统字段（id 自动生成，createdAt 自动填充） |
| `update` 参数用 `Partial<Omit<T, "id" | "createdAt">>` | 部分更新（不可改 id 和 createdAt） |
| `findByProperty` 的 `key` 用 `keyof T` | key/value 类型对齐 |
| Task 和 BugReport 共用同一套泛型代码 | 验证复用 |

```typescript
// 泛型实例化正确
const state = createInitialState<Task>();
const newState = createEntity(state, { title: "test", completed: false, priority: "high" });
// newState.items[0] 的类型应为 Task

// 类型安全——非法操作应报错
// createEntity(state, { wrong: "field" }); // ❌
// findByProperty(state, "nonexistent", 1); // ❌
```
