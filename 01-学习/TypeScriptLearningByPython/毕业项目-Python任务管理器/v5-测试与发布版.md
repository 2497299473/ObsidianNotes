## 🎯 v5 目标

这是毕业项目的最终版本——从 v1 的"能编译"到 v4 的"工程化"，v5 实现**可测试、可 CI、可发布**。对应 Python 的 `pytest` + `tox` + `twine` 全流程。

---

## 📊 全流程对比

| 步骤 | Python | TypeScript | v5 实现 |
|------|--------|-----------|---------|
| 测试框架 | `pytest` | `vitest` | 单元测试 + 覆盖率 |
| 类型检查 | `mypy` | `tsc --noEmit` | 构建前强制检查 |
| lint | `ruff` / `flake8` | `eslint` | 不强制，但可加 |
| 格式化 | `black` | `prettier` | 不强制，但可加 |
| 多环境测试 | `tox` | `vitest` matrix | Node 20/22/24 |
| CI | GitHub Actions | GitHub Actions | 相同 |
| 发布 | `twine` → PyPI | `npm publish` | 发布到 npm |
| 版本锁 | `poetry.lock` | `package-lock.json` | 自动生成 |

## ✏️ 第一步：安装 vitest

```bash
npm install -D vitest @vitest/coverage-v8
```

> **对应 Python**：`pip install pytest pytest-cov`

在 `package.json` 中添加测试脚本：

```jsonc
{
  "scripts": {
    "test": "vitest run",
    "test:watch": "vitest",
    "test:coverage": "vitest run --coverage"
  }
}
```

| npm script | Python 等价 |
|------------|-------------|
| `npm test` | `pytest` |
| `npm run test:watch` | `pytest-watch` / `ptw` |
| `npm run test:coverage` | `pytest --cov=src/` |

## ✏️ 第二步：配置 vitest

```typescript
// vitest.config.ts
import { defineConfig } from "vitest/config";

export default defineConfig({
  test: {
    globals: true,          // 全局 describe/it/expect
    environment: "node",    // Node 环境
    include: ["tests/**/*.test.ts"],
    coverage: {
      provider: "v8",
      include: ["src/**/*.ts"],
      exclude: ["src/index.ts"],  // 入口文件不需要测
      thresholds: {
        branches: 80,
        lines: 80,
      },
    },
  },
});
```

## ✏️ 第三步：写测试

### tests/repository.test.ts

```typescript
import { describe, it, expect, beforeEach } from "vitest";
import { InMemoryRepository, TaskError } from "../src/repository.js";
import type { TaskRepository } from "../src/repository.js";

describe("InMemoryRepository", () => {
  let repo: TaskRepository;

  beforeEach(() => {
    repo = new InMemoryRepository();
  });

  describe("create", () => {
    it("should create a task with auto-incremented id", async () => {
      const task = await repo.create("Test", "high");
      expect(task).toMatchObject({
        id: expect.any(Number),
        title: "Test",
        completed: false,
        priority: "high",
        createdAt: expect.any(Date),
      });
    });

    it("should reject empty title", async () => {
      await expect(repo.create("", "normal")).rejects.toThrow(TaskError);
      await expect(repo.create(" ", "normal")).rejects.toThrow(TaskError);
    });
  });

  describe("findById", () => {
    it("should return task when found", async () => {
      const created = await repo.create("Found", "normal");
      const found = await repo.findById(created.id);
      expect(found).toEqual(created);
    });

    it("should return undefined when not found", async () => {
      const found = await repo.findById(999);
      expect(found).toBeUndefined();
    });
  });

  describe("findAll", () => {
    it("should filter by status", async () => {
      const t1 = await repo.create("Task 1", "normal");
      const t2 = await repo.create("Task 2", "high");
      await repo.update(t1.id, { completed: true });

      const all = await repo.findAll("all");
      const completed = await repo.findAll("completed");
      const pending = await repo.findAll("pending");

      expect(all.length).toBe(2);
      expect(completed.length).toBe(1);
      expect(pending.length).toBe(1);
    });
  });

  describe("update", () => {
    it("should update task properties", async () => {
      const task = await repo.create("Original", "low");
      const updated = await repo.update(task.id, { title: "Updated" });
      expect(updated.title).toBe("Updated");
    });

    it("should throw for non-existent task", async () => {
      await expect(repo.update(999, { title: "Ghost" })).rejects.toThrow(TaskError);
    });
  });

  describe("delete", () => {
    it("should delete existing task", async () => {
      const task = await repo.create("Delete me", "normal");
      expect(await repo.delete(task.id)).toBe(true);
      expect(await repo.findById(task.id)).toBeUndefined();
    });

    it("should return false for non-existent task", async () => {
      expect(await repo.delete(999)).toBe(false);
    });
  });
});
```

### tests/task-manager.test.ts

```typescript
import { describe, it, expect, beforeEach } from "vitest";
import { InMemoryRepository } from "../src/repository.js";
import { TaskManager } from "../src/task-manager.js";

describe("TaskManager", () => {
  let manager: TaskManager;

  beforeEach(() => {
    manager = new TaskManager(new InMemoryRepository());
  });

  it("should add and list tasks", async () => {
    await manager.add("First", "high");
    await manager.add("Second", "normal");
    const tasks = await manager.list("all");
    expect(tasks.length).toBe(2);
  });

  it("should complete a task", async () => {
    const task = await manager.add("To complete", "normal");
    const completed = await manager.complete(task.id);
    expect(completed.completed).toBe(true);
  });

  it("should get correct stats", async () => {
    const t1 = await manager.add("One", "high");
    await manager.add("Two", "normal");
    await manager.complete(t1.id);
    const stats = await manager.getStats();
    expect(stats).toEqual({ total: 2, completed: 1, pending: 1 });
  });

  it("should handle removing all tasks", async () => {
    const t1 = await manager.add("One", "low");
    const t2 = await manager.add("Two", "low");
    await manager.remove(t1.id);
    await manager.remove(t2.id);
    const stats = await manager.getStats();
    expect(stats.total).toBe(0);
  });
});
```

> [!tip] pytest 对比表
> | pytest | vitest |
> |--------|--------|
> | `def test_something():` | `it("should ...", () => {})` |
> | `@pytest.fixture` | `beforeEach` / `beforeAll` |
> | `assert ...` | `expect(...).toBe(...)` / `expect(...).toEqual(...)` |
> | `pytest.raises(X)` | `expect(fn()).rejects.toThrow(X)` |
> | `pytest.mark.parametrize` | `it.each` |
> | `pytest --cov` | `vitest run --coverage` |
> | `conftest.py` | `vitest.config.ts` 的 `setupFiles` |

## ✏️ 第四步：CI 配置

### .github/workflows/ci.yml

```yaml
name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        node-version: [18, 20, 22]

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js ${{ matrix.node-version }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}

      - name: Install dependencies
        run: npm ci

      - name: Type check
        run: npm run typecheck

      - name: Run tests
        run: npm run test:coverage

      - name: Build
        run: npm run build

  publish:
    needs: test
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          registry-url: "https://registry.npmjs.org"
      - run: npm ci
      - run: npm run build
      - run: npm publish
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
```

> [!tip] GitHub Actions 是通用的
> CI 配置在 Python 和 TypeScript 中几乎一样——只是安装步骤从 `pip install` 改为 `npm ci`，测试命令从 `pytest` 改为 `vitest run`。

## ✏️ 第五步：发布到 npm

```bash
# 1. 确保 tests 通过 + 覆盖率达标
npm test
npm run test:coverage

# 2. 类型检查
npm run typecheck

# 3. 构建
npm run build

# 4. 检查构建产物
ls dist/
# 期望：index.js, index.d.ts, types.js, types.d.ts, ...

# 5. 发布
npm publish --access public
```

### package.json 发布字段

```jsonc
{
  "name": "@your-scope/task-manager",
  "version": "1.0.0",
  "main": "./dist/index.js",
  "exports": {
    ".": {
      "import": "./dist/index.js",
      "types": "./dist/index.d.ts"
    }
  },
  "files": ["dist"],
  "publishConfig": {
    "access": "public"
  }
}
```

| npm 发布字段 | Python 等价 | 说明 |
|-------------|-------------|------|
| `name` | `[project] name` | 包名 |
| `version` | `[project] version` | 版本号 |
| `files` | `MANIFEST.in` | 发布时包含的文件 |
| `publishConfig.access` | 无（PyPI 默认公开） | npm 默认私有（付费） |
| `exports` | 无 | 控制子路径导出 |
