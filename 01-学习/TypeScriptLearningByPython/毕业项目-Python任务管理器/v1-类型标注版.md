## 🎯 v1 目标

把一段 **Python 任务管理器代码**迁移为 TypeScript，**只用类型标注，不改逻辑**。这是"能不能用 TS"的判定。

> [!important] v1 的核心约束
> - ✅ 只加类型标注（interface/type/函数参数/返回值）
> - ✅ 可以用字面量联合替代 Python 的字符串参数
> - ❌ 不改逻辑、不重构、不用泛型
> - ❌ 不用条件类型、映射类型等高级特性

## 📋 原始 Python 代码

```python
from dataclasses import dataclass
from typing import Literal

Priority = Literal["low", "normal", "high"]
Filter = Literal["all", "completed", "pending"]

@dataclass
class Task:
    id: int
    title: str
    completed: bool = False
    priority: Priority = "normal"
    created_at: str = ""  # ISO 格式

tasks: list[Task] = []
next_id = 1

def add_task(title: str, priority: Priority = "normal") -> Task:
    global next_id
    task = Task(id=next_id, title=title, priority=priority)
    next_id += 1
    tasks.append(task)
    return task

def complete_task(task_id: int) -> bool:
    for task in tasks:
        if task.id == task_id:
            task.completed = True
            return True
    return False

def remove_task(task_id: int) -> bool:
    global tasks
    original_len = len(tasks)
    tasks = [t for t in tasks if t.id != task_id]
    return len(tasks) != original_len

def list_tasks(filter_status: Filter = "all") -> list[Task]:
    if filter_status == "completed":
        return [t for t in tasks if t.completed]
    if filter_status == "pending":
        return [t for t in tasks if not t.completed]
    return tasks

def get_stats() -> dict[str, int]:
    total = len(tasks)
    completed = len([t for t in tasks if t.completed])
    return {"total": total, "completed": completed, "pending": total - completed}

# 使用
add_task("学习 TypeScript", "high")
add_task("写毕业项目", "high")
complete_task(1)
print(get_stats())
```

```typescript
// types.ts
type Priority = "low" | "normal" | "high";
type Filter = "all" | "completed" | "pending";

interface Task {
  id: number;
  title: string;
  completed: boolean;
  priority: Priority;
  createdAt: Date;
}

interface Stats {
  total: number;
  completed: number;
  pending: number;
}

// task-manager.ts
const tasks: Task[] = [];
let nextId = 1;

function addTask(title: string, priority: Priority = "normal"): Task {
  const task: Task = {
    id: nextId++,
    title,
    completed: false,
    priority,
    createdAt: new Date(),
  };
  tasks.push(task);
  return task;
}

function completeTask(taskId: number): boolean {
  const task = tasks.find((t) => t.id === taskId);
  if (task) {
    task.completed = true;
    return true;
  }
  return false;
}

function removeTask(taskId: number): boolean {
  const idx = tasks.findIndex((t) => t.id === taskId);
  if (idx !== -1) {
    tasks.splice(idx, 1);
    return true;
  }
  return false;
}

function listTasks(filter: Filter = "all"): Task[] {
  if (filter === "completed") return tasks.filter((t) => t.completed);
  if (filter === "pending") return tasks.filter((t) => !t.completed);
  return tasks;
}

function getStats(): Stats {
  const total = tasks.length;
  const completed = tasks.filter((t) => t.completed).length;
  return { total, completed, pending: total - completed };
}

// 使用
addTask("学习 TypeScript", "high");
addTask("写毕业项目", "high");
completeTask(1);
console.log(getStats());
```

## 🐍 Python→TS 迁移对比

| 维度 | Python | TypeScript |
|------|--------|-----------|
| 数据类定义 | `@dataclass` + 类型注解 | `interface` 定义形状 |
| 枚举/字面量 | `Literal["low", "normal", "high"]` | `"low" \| "normal" \| "high"` |
| 列表推导式 | `[t for t in tasks if ...]` | `tasks.filter(t => ...)` |
| 查找第一个 | `next((t for t in ...), None)` | `tasks.find(t => ...)` |
| 全局变量 | `global next_id` | `let nextId`（模块级，无需 global） |
| 默认参数 | `priority = "normal"` | `priority: Priority = "normal"` |
| 类型注解位置 | 变量后 `: type` | 变量后 `: type`（一致） |
| 返回值 | `-> Type` | `: Type`（在参数后） |
| 长度 | `len(tasks)` | `tasks.length` |

> [!tip] Python 开发者注意
> TS 的 `interface` 是纯类型定义，编译后完全擦除；Python 的 `@dataclass` 是运行时类。TS 的 `filter` 返回新数组，Python 的列表推导式也是——行为一致，语法不同。
