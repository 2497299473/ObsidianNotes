## 🎯 v3 目标

在 v2 的基础上做两个改造：
1. **异步化**——所有 Repository 方法改为 `async`，返回 `Promise<T>`
2. **类型安全错误处理**——用自定义错误类 + 可辨识联合 `Result` 替代 Python 的 `try/except`

> [!tip] Python 对照
> Python 用 `try/except` 处理错误，异常类型在运行时才知道。TS 的可辨识联合 `Result` 可以在**编译时**确保所有错误分支都被处理——这是 Python 完全做不到的。

## ✏️ 你的任务

### 第一步：定义自定义错误类型

```typescript
class TaskNotFoundError extends Error {
  constructor(public taskId: number) {
    super(`Task ${taskId} not found`);
    this.name = "TaskNotFoundError";
  }
}

class ValidationError extends Error {
  constructor(public field: string, public reason: string) {
    super(`Validation failed: ${field} - ${reason}`);
    this.name = "ValidationError";
  }
}

type AppError = TaskNotFoundError | ValidationError;
```

### 第二步：将 Repository 接口改为异步

```typescript
interface AsyncRepository<T extends Identifiable> {
  findAll(): Promise<T[]>;
  findById(id: number): Promise<T | undefined>;
  create(item: Omit<T, "id">): Promise<T>;
  update(id: number, updates: Partial<T>): Promise<T | undefined>;
  delete(id: number): Promise<boolean>;
}
```

### 第三步：实现类型守卫

```typescript
function isTaskNotFoundError(error: unknown): error is TaskNotFoundError {
  return error instanceof TaskNotFoundError;
}

function isValidationError(error: unknown): error is ValidationError {
  return error instanceof ValidationError;
}
```

### 第四步：用可辨识联合改写结果

```typescript
// Python 完全没有的 TS 独有模式！
type Result<T, E = AppError> =
  | { ok: true; value: T }
  | { ok: false; error: E };

function success<T>(value: T): Result<T, never> {
  return { ok: true, value };
}

function failure<E>(error: E): Result<never, E> {
  return { ok: false, error };
}
```

### 第五步：重构 TaskManager 使用 Result

```typescript
class TaskManager {
  constructor(private repo: AsyncRepository<Task>) {}

  async add(title: string, priority: Priority = "normal"): Promise<Result<Task>> {
    if (!title.trim()) {
      return failure(new ValidationError("title", "cannot be empty"));
    }
    const task = await this.repo.create({
      title: title.trim(),
      completed: false,
      priority,
      createdAt: new Date(),
    });
    return success(task);
  }

  async complete(id: number): Promise<Result<Task>> {
    const task = await this.repo.update(id, { completed: true });
    if (!task) return failure(new TaskNotFoundError(id));
    return success(task);
  }

// === 类型定义 ===
type Priority = "low" | "normal" | "high";
type Filter = "all" | "completed" | "pending";

interface Identifiable {
  id: number;
}

interface Task extends Identifiable {
  id: number;
  title: string;
  completed: boolean;
  priority: Priority;
  createdAt: Date;
}

interface Stats {
  total: number;
  completed: number;
  pending: number;
}

// === 自定义错误类型 ===
class TaskNotFoundError extends Error {
  constructor(public taskId: number) {
    super(`Task ${taskId} not found`);
    this.name = "TaskNotFoundError";
  }
}

class ValidationError extends Error {
  constructor(public field: string, public reason: string) {
    super(`Validation failed: ${field} - ${reason}`);
    this.name = "ValidationError";
  }
}

type AppError = TaskNotFoundError | ValidationError;

// === 类型守卫 ===
function isTaskNotFoundError(error: unknown): error is TaskNotFoundError {
  return error instanceof TaskNotFoundError;
}

function isValidationError(error: unknown): error is ValidationError {
  return error instanceof ValidationError;
}

// === 可辨识联合 Result ===
type Result<T, E = AppError> =
  | { ok: true; value: T }
  | { ok: false; error: E };

function success<T>(value: T): Result<T, never> {
  return { ok: true, value };
}

function failure<E>(error: E): Result<never, E> {
  return { ok: false, error };
}

// === 异步 Repository 接口 ===
interface AsyncRepository<T extends Identifiable> {
  findAll(): Promise<T[]>;
  findById(id: number): Promise<T | undefined>;
  create(item: Omit<T, "id">): Promise<T>;
  update(id: number, updates: Partial<T>): Promise<T | undefined>;
  delete(id: number): Promise<boolean>;
}

// === 异步内存仓储（模拟异步数据库）===
class AsyncMemoryRepository<T extends Identifiable> implements AsyncRepository<T> {
  private items: T[] = [];
  private nextId = 1;

  private async delay<T>(value: T): Promise<T> {
    return new Promise((resolve) => setTimeout(() => resolve(value), 10));
  }

  async findAll(): Promise<T[]> {
    await this.delay(null);
    return [...this.items];
  }

  async findById(id: number): Promise<T | undefined> {
    await this.delay(null);
    return this.items.find((item) => item.id === id);
  }

  async create(data: Omit<T, "id">): Promise<T> {
    await this.delay(null);
    const item = { ...data, id: this.nextId++ } as T;
    this.items.push(item);
    return item;
  }

  async update(id: number, updates: Partial<T>): Promise<T | undefined> {
    await this.delay(null);
    const item = this.items.find((i) => i.id === id);
    if (!item) return undefined;
    Object.assign(item, updates);
    return item;
  }

  async delete(id: number): Promise<boolean> {
    await this.delay(null);
    const idx = this.items.findIndex((i) => i.id === id);
    if (idx === -1) return false;
    this.items.splice(idx, 1);
    return true;
  }
}

// === 服务层：返回 Result 而非 throw ===
class TaskManager {
  constructor(private repo: AsyncRepository<Task>) {}

  async add(title: string, priority: Priority = "normal"): Promise<Result<Task>> {
    if (!title.trim()) {
      return failure(new ValidationError("title", "cannot be empty"));
    }
    const task = await this.repo.create({
      title: title.trim(),
      completed: false,
      priority,
      createdAt: new Date(),
    });
    return success(task);
  }

  async complete(id: number): Promise<Result<Task>> {
    const task = await this.repo.update(id, { completed: true });
    if (!task) return failure(new TaskNotFoundError(id));
    return success(task);
  }

  async remove(id: number): Promise<Result<boolean>> {
    const deleted = await this.repo.delete(id);
    if (!deleted) return failure(new TaskNotFoundError(id));
    return success(true);
  }

  async list(filter: Filter = "all"): Promise<Task[]> {
    const all = await this.repo.findAll();
    if (filter === "completed") return all.filter((t) => t.completed);
    if (filter === "pending") return all.filter((t) => !t.completed);
    return all;
  }

  async getStats(): Promise<Stats> {
    const all = await this.repo.findAll();
    const completed = all.filter((t) => t.completed).length;
    return { total: all.length, completed, pending: all.length - completed };
  }
}

// === 使用 ===
async function main() {
  const repo = new AsyncMemoryRepository<Task>();
  const manager = new TaskManager(repo);

  const result = await manager.add("学习 TypeScript", "high");
  if (result.ok) {
    console.log(`✅ Created: ${result.value.title}`);
  } else {
    if (isValidationError(result.error)) {
      console.log(`⚠️ ${result.error.field}: ${result.error.reason}`);
    }
  }

  const completed = await manager.complete(1);
  if (!completed.ok) {
    if (isTaskNotFoundError(completed.error)) {
      console.log(`❌ Task ${completed.error.taskId} not found`);
    }
  }

  console.log(await manager.getStats());
}

main();
```

## 🐍 Python→TS 错误处理对比

| 维度 | Python | TypeScript |
|------|--------|-----------|
| 错误抛出 | `raise NotFoundError(id)` | `return failure(new NotFoundError(id))` |
| 错误捕获 | `try/except NotFoundError` | `if (!result.ok) { ... }` |
| 编译时检查 | 无（运行时才发现遗漏分支） | 有（穷尽性检查） |
| 类型守卫 | `isinstance(e, NotFoundError)` | `isTaskNotFoundError(e)`（`e is T`） |
| 错误类型联合 | 无原生 | `type AppError = A \| B`（可辨识联合） |

> [!important] Result 模式 vs try/catch
> Python 的 `try/except` 是运行时机制，无法在编译时保证所有错误分支被处理。TS 的 `Result<T, E>` 是**编译时类型安全**的——如果你漏了某个错误分支，编译器会报错。
