## 毕业项目 v4：事务与并发控制

**改造维度**：事务模型 + 并发写入
**MySQL 锚点**：隔离级别 + 行锁 → MVCC + 死锁处理
**验收标准**：并发场景零死锁 + 正确隔离 + 队列消费无阻塞

### 目标

在 v3 的查询优化基础上，加入事务控制和并发处理能力，模拟真实生产场景。

### 任务清单

- [ ] 验证 MVCC 快照隔离（两个会话对比）
- [ ] 测试 PG 的 Repeatable Read 防幻读
- [ ] 验证 DDL 事务可回滚（PG 独有）
- [ ] 实现安全库存扣减（SELECT FOR UPDATE + EXCEPTION）
- [ ] 用 SKIP LOCKED 实现高并发队列消费
- [ ] 构造死锁并观察检测过程
- [ ] 用 Advisory Lock 实现应用层分布式锁
- [ ] 观察长事务对 VACUUM 的影响

-- ============================================
-- Task 1：MVCC 快照隔离验证（需两个会话）
-- ============================================
-- 会话 A：
BEGIN ISOLATION LEVEL REPEATABLE READ;
SELECT COUNT(*) FROM orders WHERE status = 'pending';
-- 记录结果，比如 25000

-- 会话 B（在另一个连接中）：
UPDATE orders SET status = 'paid' WHERE status = 'pending' AND id < 100;
COMMIT;

-- 会话 A（会话 B 提交后）：
SELECT COUNT(*) FROM orders WHERE status = 'pending';
-- PG 结果：25000（仍然看到旧值，快照隔离！）
COMMIT;

-- ============================================
-- Task 2：隔离级别测试（PG RR 防幻读）
-- ============================================
SHOW transaction_isolation;  -- read committed（PG 默认）

-- 测试 RR 普通快照读的防幻读表现（PG vs MySQL）
-- 会话 A：
BEGIN ISOLATION LEVEL REPEATABLE READ;
SELECT COUNT(*) FROM orders WHERE total_amount > 500;

-- 会话 B：
INSERT INTO orders (user_id, status, total_amount) VALUES (1, 'paid', 999);
COMMIT;

-- 会话 A：
SELECT COUNT(*) FROM orders WHERE total_amount > 500;
-- PG RR：看不到新插入的行（无锁快照隔离，防幻读）
-- MySQL RR：普通快照读同样看不到新插入的行；幻读出现在锁定读场景
--           （SELECT ... FOR UPDATE/SHARE），next-key 锁（记录锁+间隙锁）
--           正是 MySQL 在锁定读下防幻读的机制
COMMIT;

-- Task 4：安全库存扣减函数
-- ============================================
-- 添加库存表
CREATE TABLE IF NOT EXISTS inventory (
    product_id INT PRIMARY KEY REFERENCES products(id),
    stock INT NOT NULL DEFAULT 0 CHECK (stock >= 0)
);

INSERT INTO inventory (product_id, stock)
SELECT id, stock FROM products WHERE id <= 3
ON CONFLICT (product_id) DO NOTHING;

-- 安全扣减函数（FOR UPDATE + EXCEPTION）
CREATE OR REPLACE FUNCTION deduct_stock(
    p_product_id INT,
    p_qty INT
) RETURNS TEXT AS $$
DECLARE
    v_stock INT;
    v_name VARCHAR;
BEGIN
    -- 锁定行（FOR UPDATE 防止并发超卖）
    SELECT name, stock INTO v_name, v_stock
    FROM products WHERE id = p_product_id FOR UPDATE;

    IF v_stock IS NULL THEN
        RETURN 'Product not found';
    END IF;

    IF v_stock < p_qty THEN
        RETURN format('Insufficient stock: %s has %s, need %s', v_name, v_stock, p_qty);
    END IF;

    UPDATE products SET stock = stock - p_qty WHERE id = p_product_id;
    RETURN format('Success: deducted %s from %s, remaining %s', p_qty, v_name, v_stock - p_qty);

EXCEPTION
    WHEN deadlock_detected THEN
        RETURN 'Deadlock detected, please retry';
    WHEN OTHERS THEN
        RETURN 'Error: ' || SQLERRM;
END;
$$ LANGUAGE plpgsql;

-- 测试
SELECT deduct_stock(1, 5);     -- Success
SELECT deduct_stock(1, 99999); -- Insufficient stock

-- ============================================
-- Task 5：SKIP LOCKED 队列消费
-- ============================================
CREATE TABLE IF NOT EXISTS task_queue (
    id SERIAL PRIMARY KEY,
    task_type VARCHAR(50) NOT NULL,
    payload JSONB DEFAULT '{}',
    status VARCHAR(20) DEFAULT 'pending',
    worker_id VARCHAR(50),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 部分索引：只索引 pending 任务
CREATE INDEX IF NOT EXISTS idx_tasks_pending ON task_queue (created_at)
WHERE status = 'pending';

-- 插入任务
INSERT INTO task_queue (task_type, payload)
SELECT 'email_send', jsonb_build_object('to', 'user' || n || '@test.com')
FROM generate_series(1, 100) AS n;

-- 消费者函数（SKIP LOCKED 不阻塞）
CREATE OR REPLACE FUNCTION consume_tasks(
    p_worker_id TEXT,
    p_limit INT DEFAULT 5
) RETURNS TABLE (id INT, task_type TEXT, payload JSONB) AS $$
BEGIN
    RETURN QUERY
    UPDATE task_queue
    SET status = 'processing', worker_id = p_worker_id
    WHERE id IN (
        SELECT id FROM task_queue
        WHERE status = 'pending'
        ORDER BY created_at
        LIMIT p_limit
        FOR UPDATE SKIP LOCKED  -- 跳过被锁定的行，不阻塞
    )
    RETURNING task_queue.id, task_queue.task_type, task_queue.payload;
END;
$$ LANGUAGE plpgsql;

-- 两个消费者并发消费（互不阻塞、不重复）
SELECT * FROM consume_tasks('worker-1', 10);
SELECT * FROM consume_tasks('worker-2', 10);

-- 标记完成
UPDATE task_queue SET status = 'completed' WHERE status = 'processing';

-- ============================================
-- Task 7：Advisory Lock 分布式锁
-- ============================================
CREATE OR REPLACE FUNCTION run_scheduled_task(p_task_name TEXT)
RETURNS TEXT AS $$
DECLARE
    v_lock_key BIGINT;
BEGIN
    -- 用任务名哈希生成锁键
    v_lock_key := abs(hashtext(p_task_name))::BIGINT;

    -- 尝试获取锁（非阻塞）
    IF NOT pg_try_advisory_lock(v_lock_key) THEN
        RETURN 'Task is already running elsewhere';
    END IF;

    -- 执行业务逻辑
    UPDATE products SET is_active = false WHERE stock = 0;
    PERFORM pg_sleep(2);  -- 模拟耗时操作

    -- 释放锁
    PERFORM pg_advisory_unlock(v_lock_key);
    RETURN 'Task completed: ' || p_task_name;
END;
$$ LANGUAGE plpgsql;

-- 测试
SELECT run_scheduled_task('daily_report');
-- 在另一个会话同时执行会返回 "already running"
