## 毕业项目 v5：存储过程与触发器迁移

**改造维度**：存储过程 + 触发器 + 分区表 + 扩展
**MySQL 锚点**：PROCEDURE → PL/pgSQL 函数
**验收标准**：PL/pgSQL 等价实现 + 触发器对齐 + 分区裁剪验证 + 扩展可用

-- ============================================
-- Task 1：存储过程迁移为 PL/pgSQL 函数
-- ============================================
-- 原 MySQL 存储过程：
-- CREATE PROCEDURE get_user_order_summary(IN p_user_id INT)
-- BEGIN
--     SELECT u.name, COUNT(o.id), SUM(o.total_amount)
--     FROM users u LEFT JOIN orders o ON u.id = o.user_id
--     WHERE u.id = p_user_id GROUP BY u.id, u.name;
-- END;

-- PL/pgSQL 函数版本（RETURNS TABLE）
CREATE OR REPLACE FUNCTION get_user_order_summary(p_user_id INT)
RETURNS TABLE (
    user_name VARCHAR,
    order_count BIGINT,
    total_amount NUMERIC
) AS $$
BEGIN
    RETURN QUERY
    SELECT u.name, COUNT(o.id), COALESCE(SUM(o.total_amount), 0)
    FROM users u
    LEFT JOIN orders o ON u.id = o.user_id
    WHERE u.id = p_user_id
    GROUP BY u.id, u.name;
END;
$$ LANGUAGE plpgsql STABLE;

SELECT * FROM get_user_order_summary(1);

-- ============================================
-- Task 2：批量更新 PROCEDURE（PG 11+ 事务控制）
-- ============================================
CREATE OR REPLACE PROCEDURE bulk_update_order_status(
    p_old_status order_status,
    p_new_status order_status
) LANGUAGE plpgsql AS $$
DECLARE
    v_count INT;
BEGIN
    UPDATE orders SET status = p_new_status
    WHERE status = p_old_status
      AND created_at < NOW() - INTERVAL '7 days';

    GET DIAGNOSTICS v_count = ROW_COUNT;
    RAISE NOTICE 'Updated % orders from % to %', v_count, p_old_status, p_new_status;
END;
$$;

-- 调用（PROCEDURE 用 CALL）
CALL bulk_update_order_status('pending', 'cancelled');

-- ============================================
-- Task 3：审计触发器（通用函数 + 多表复用）
-- ============================================
CREATE TABLE IF NOT EXISTS audit_log (
    id BIGSERIAL PRIMARY KEY,
    table_name TEXT NOT NULL,
    operation TEXT NOT NULL,
    old_data JSONB,
    new_data JSONB,
    changed_by TEXT DEFAULT current_user,
    changed_at TIMESTAMPTZ DEFAULT NOW()
);

-- 通用审计触发器函数（可被任意表复用）
CREATE OR REPLACE FUNCTION audit_trigger_func()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO audit_log (table_name, operation, new_data)
        VALUES (TG_TABLE_NAME, 'INSERT', to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO audit_log (table_name, operation, old_data, new_data)
        VALUES (TG_TABLE_NAME, 'UPDATE', to_jsonb(OLD), to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO audit_log (table_name, operation, old_data)
        VALUES (TG_TABLE_NAME, 'DELETE', to_jsonb(OLD));
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- 绑定审计触发器到 products 和 orders 表
CREATE TRIGGER trg_audit_products
    AFTER INSERT OR UPDATE OR DELETE ON products
    FOR EACH ROW EXECUTE FUNCTION audit_trigger_func();

CREATE TRIGGER trg_audit_orders
    AFTER INSERT OR UPDATE OR DELETE ON orders
    FOR EACH ROW EXECUTE FUNCTION audit_trigger_func();

-- 测试审计
INSERT INTO products (name, specs, price, stock, tags)
VALUES ('Test Product', '{"brand":"Test"}', 100, 10, ARRAY['test']);
UPDATE products SET price = 150 WHERE name = 'Test Product';
DELETE FROM products WHERE name = 'Test Product';

SELECT id, table_name, operation, changed_at
FROM audit_log ORDER BY id DESC LIMIT 5;

-- ============================================
-- Task 4：物化视图 + CONCURRENTLY 刷新
-- ============================================
CREATE MATERIALIZED VIEW mv_daily_sales AS
SELECT
    DATE_TRUNC('day', created_at)::DATE AS stat_date,
    status,
    COUNT(*) AS order_count,
    SUM(total_amount) AS revenue,
    AVG(total_amount) AS avg_order_value
FROM orders
GROUP BY DATE_TRUNC('day', created_at), status;

-- 唯一索引（支持 CONCURRENTLY 刷新）
CREATE UNIQUE INDEX idx_mv_daily_sales ON mv_daily_sales (stat_date, status);

-- 查询（毫秒级响应）
SELECT * FROM mv_daily_sales
WHERE stat_date >= CURRENT_DATE - INTERVAL '7 days'
ORDER BY stat_date DESC, revenue DESC;

-- 并发刷新（不阻塞查询）
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_daily_sales;

-- ============================================
-- Task 5：分区表 + 验证分区裁剪
-- ============================================
CREATE TABLE server_logs (
    id BIGSERIAL,  -- 分区父表在 PG 17 前不支持 IDENTITY，基线 postgres:16 用 BIGSERIAL
    log_time TIMESTAMPTZ NOT NULL,
    level VARCHAR(10) NOT NULL,
    service VARCHAR(50),
    message TEXT,
    context JSONB DEFAULT '{}'
) PARTITION BY RANGE (log_time);

-- 创建分区
CREATE TABLE server_logs_2026_07 PARTITION OF server_logs
    FOR VALUES FROM ('2026-07-01') TO ('2026-08-01');
CREATE TABLE server_logs_2026_08 PARTITION OF server_logs
    FOR VALUES FROM ('2026-08-01') TO ('2026-09-01');

-- BRIN 索引（时序大表最佳选择）
CREATE INDEX idx_logs_time_brin ON server_logs USING BRIN (log_time);
-- JSONB GIN 索引
CREATE INDEX idx_logs_context ON server_logs USING GIN (context);

-- 插入测试数据
INSERT INTO server_logs (log_time, level, service, message)
SELECT
    NOW() - (n || ' minutes')::INTERVAL,
    CASE WHEN n % 100 = 0 THEN 'ERROR'
         WHEN n % 10 = 0 THEN 'WARN'
         ELSE 'INFO' END,
    'api-server',
    'Log message #' || n
FROM generate_series(1, 10000) AS n;

-- 验证分区裁剪
EXPLAIN (ANALYZE)
SELECT * FROM server_logs
WHERE log_time >= '2026-07-20' AND log_time < '2026-07-25';
-- 预期：只扫描 server_logs_2026_07（其他分区被裁剪）

-- ============================================
-- Task 6：扩展生态启用
-- ============================================
-- 1. pg_stat_statements（SQL 性能分析）
CREATE EXTENSION IF NOT EXISTS pg_stat_statements;

SELECT LEFT(query, 80) AS query, calls,
       ROUND(mean_exec_time::numeric, 2) AS avg_ms,
       rows
FROM pg_stat_statements
WHERE query NOT ILIKE '%pg_stat_statements%'
ORDER BY total_exec_time DESC LIMIT 5;

-- 2. pg_trgm（模糊搜索）
CREATE EXTENSION IF NOT EXISTS pg_trgm;

CREATE INDEX idx_products_name_trgm ON products USING GIN (name gin_trgm_ops);

SELECT name, similarity(name, 'iPhone') AS sim
FROM products
WHERE name % 'iPhone'
ORDER BY sim DESC LIMIT 5;

-- 3. uuid-ossp（UUID 生成）
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
SELECT uuid_generate_v4();

-- ============================================
-- Task 7：动态 SQL + 多态函数
-- ============================================
-- 动态 SQL（按表名统计，用 format 防注入）
CREATE OR REPLACE FUNCTION count_by_table(p_table TEXT)
RETURNS BIGINT AS $$
DECLARE v_count BIGINT;
BEGIN
    EXECUTE format('SELECT COUNT(*) FROM %I', p_table) INTO v_count;
    RETURN v_count;
END;
$$ LANGUAGE plpgsql;

SELECT count_by_table('orders');
SELECT count_by_table('products');

-- 多态函数（同名不同参数类型，PG 独有）
CREATE OR REPLACE FUNCTION search_products(p_keyword TEXT)
RETURNS TABLE (id INT, name TEXT, price NUMERIC) AS $$
BEGIN
    RETURN QUERY
    SELECT p.id, p.name, p.price
    FROM products p
    WHERE p.name ILIKE '%' || p_keyword || '%';
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION search_products(p_min_price NUMERIC, p_max_price NUMERIC)
RETURNS TABLE (id INT, name TEXT, price NUMERIC) AS $$
BEGIN
    RETURN QUERY
    SELECT p.id, p.name, p.price
    FROM products p
    WHERE p.price BETWEEN p_min_price AND p_max_price;
END;
$$ LANGUAGE plpgsql;

-- 调用（PG 根据参数类型自动选择）
SELECT * FROM search_products('iPhone');       -- 按关键字
SELECT * FROM search_products(100.0, 500.0);   -- 按价格范围
