# v2-多数据结构组合版

> 📍 **前置**：[[v1-单机缓存服务版]] → 本版本 → [[v3-持久化与AOF版]]

## 🎯 本版本目标

用 Hash/ZSet/Set/List 实现排行榜、购物车、标签等业务 API，实现：**多结构组合业务场景**。

---

## 💻 核心实现

```python
# 排行榜（ZSet）
@app.route('/leaderboard/<game>', methods=['POST'])
def add_score(game):
    data = request.get_json()
    r.zincrby(f"rank:{game}", data['score'], data['player'])
    return jsonify({"status": "ok"})

@app.route('/leaderboard/<game>/top', methods=['GET'])
def top(game):
    n = request.args.get('n', 10, type=int)
    result = r.zrevrange(f"rank:{game}", 0, n-1, withscores=True)
    return jsonify([{"player": p, "score": s} for p, s in result])

# 购物车（Hash）
@app.route('/cart/<user_id>', methods=['POST'])
def add_cart(user_id):
    data = request.get_json()
    r.hincrby(f"cart:{user_id}", data['item_id'], data['quantity'])
    return jsonify(r.hgetall(f"cart:{user_id}"))

# 共同标签（Set）
@app.route('/tags/<user_id>/common', methods=['GET'])
def common_tags(user_id):
    other = request.args.get('other')
    common = r.sinter(f"tags:{user_id}", f"tags:{other}")
    return jsonify(list(common))

# 最新消息（List）
@app.route('/timeline/<user_id>', methods=['POST'])
def add_timeline(user_id):
    data = request.get_json()
    r.lpush(f"timeline:{user_id}", data['msg'])
    r.ltrim(f"timeline:{user_id}", 0, 999)  # 只保留1000条
    return jsonify({"status": "ok"})

@app.route('/timeline/<user_id>', methods=['GET'])
def get_timeline(user_id):
    msgs = r.lrange(f"timeline:{user_id}", 0, 9)
    return jsonify(msgs)
```
