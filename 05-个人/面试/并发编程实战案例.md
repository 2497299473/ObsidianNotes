## 二、`ProcessPoolExecutor` — CPU 密集型首选

```python
from concurrent.futures import ProcessPoolExecutor, as_completed
import numpy as np
import time


# ── 场景：批量仿真计算（每个工况独立，可并行） ──

def simulate_single(condition: dict) -> dict:
    """单个工况的仿真计算——纯 CPU 密集型"""
    evap_temp = condition["evap_temp"]
    cond_temp = condition["cond_temp"]

    # 模拟热力学计算（实际项目中是 NumPy/SciPy 大量运算）
    cop = 3.0 + 0.05 * evap_temp - 0.03 * cond_temp
    power = 2.5 + np.sin(evap_temp / 10)

    # 模拟计算耗时
    time.sleep(0.1)  # 实际项目中这里没有 sleep，全是 CPU

    return {
        "evap_temp": evap_temp,
        "cond_temp": cond_temp,
        "cop": round(cop, 2),
        "power_kw": round(power, 2),
    }


def run_batch_simulation():
    # 构造 20 个工况
    conditions = [
        {"evap_temp": t, "cond_temp": 45}
        for t in range(5, 25)
    ]

    # 方式一：map —— 保持顺序，所有任务完成后返回
    with ProcessPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(simulate_single, conditions))
        # results[i] 对应 conditions[i]，顺序不变

    # 方式二：submit + as_completed —— 谁先完成先处理谁
    with ProcessPoolExecutor(max_workers=4) as executor:
        futures = {
            executor.submit(simulate_single, cond): cond
            for cond in conditions
        }
        for future in as_completed(futures):
            cond = futures[future]
            result = future.result()
            print(f"完成: 蒸发温度 {cond['evap_temp']}°C → COP={result['cop']}")

    return results


# ── 场景：带超时的执行 ──
def run_with_timeout():
    with ProcessPoolExecutor(max_workers=4) as executor:
        future = executor.submit(simulate_single, {"evap_temp": 10, "cond_temp": 45})
        try:
            result = future.result(timeout=30)  # 30 秒超时
        except TimeoutError:
            print("仿真超时，可能参数不合理导致不收敛")
            future.cancel()

        # 批量超时
        futures = [executor.submit(simulate_single, c) for c in conditions]
        for f in as_completed(futures, timeout=60):
            try:
                f.result()
            except Exception as e:
                print(f"任务失败: {e}")
```

## 三、`ThreadPoolExecutor` — I/O 密集型首选

```python
from concurrent.futures import ThreadPoolExecutor, as_completed
import requests
import time


# ── 场景：并发下载多个文件 ──
def download_file(url: str, dest: str):
    resp = requests.get(url, timeout=30)
    with open(dest, "wb") as f:
        f.write(resp.content)
    return {"url": url, "size": len(resp.content), "status": resp.status_code}


def batch_download(urls: list, dest_dir: str):
    with ThreadPoolExecutor(max_workers=5) as executor:
        futures = [
            executor.submit(download_file, url, f"{dest_dir}/file_{i}.bin")
            for i, url in enumerate(urls)
        ]
        for future in as_completed(futures):
            result = future.result()
            print(f"下载完成: {result['url'][:50]}... ({result['size']} bytes)")


# ── 场景：并发数据库查询 ──
def batch_db_query(queries: list, db_config: dict):
    """并发执行多个只读查询，合并结果"""
    import psycopg2

    def execute_query(sql: str):
        conn = psycopg2.connect(**db_config)
        try:
            with conn.cursor() as cur:
                cur.execute(sql)
                return cur.fetchall()
        finally:
            conn.close()

    with ThreadPoolExecutor(max_workers=len(queries)) as executor:
        future_to_query = {
            executor.submit(execute_query, q): q
            for q in queries
        }
        results = {}
        for future in as_completed(future_to_query):
            query = future_to_query[future]
            results[query] = future.result()
    return results
```

## 四、`multiprocessing.Pool` — 老牌 API，功能更全

```python
from multiprocessing import Pool, cpu_count
import numpy as np


# ── 场景：大规模数值计算 ──
def heavy_compute(params):
    """纯 CPU 计算：矩阵乘法、FFT、卷积等"""
    size, seed = params
    np.random.seed(seed)
    a = np.random.randn(size, size)
    b = np.random.randn(size, size)
    c = np.dot(a, b)            # 矩阵乘法，CPU 密集
    return np.trace(c)          # 返回迹


# 方式一：map —— 阻塞直到所有完成
def run_with_pool_map():
    tasks = [(1000, i) for i in range(8)]  # 8 个 1000×1000 矩阵乘法

    # 不指定 processes 则默认 cpu_count()
    with Pool(processes=4) as pool:
        results = pool.map(heavy_compute, tasks)
    return results


# 方式二：imap_unordered —— 流式处理，不阻塞
def run_with_imap():
    tasks = [(1000, i) for i in range(8)]

    with Pool(processes=4) as pool:
        for i, result in enumerate(pool.imap_unordered(heavy_compute, tasks)):
            print(f"第 {i+1} 个结果: {result}")
            # 每完成一个就处理一个，不等待全部完成


# 方式三：apply_async —— 异步提交 + 回调
def on_result(result):
    print(f"计算结果: {result}")

def on_error(error):
    print(f"计算出错: {error}")

def run_with_async():
    pool = Pool(processes=4)
    for i in range(8):
        pool.apply_async(
            heavy_compute,
            args=((1000, i),),
            callback=on_result,
            error_callback=on_error,
        )
    pool.close()   # 不再接受新任务
    pool.join()    # 等待所有任务完成


# 方式四：starmap —— 参数是 tuple 的解包版 map
def multi_arg_func(a, b, c):
    return a + b * c

def run_starmap():
    args = [(1, 2, 3), (4, 5, 6), (7, 8, 9)]
    with Pool(3) as pool:
        results = pool.starmap(multi_arg_func, args)
    # results = [7, 34, 79]
```

### `multiprocessing.Pool` vs `ProcessPoolExecutor`

| | `Pool` | `ProcessPoolExecutor` |
|------|--------|---------------------|
| **API 风格** | 函数式 | 面向对象（`Future`） |
| **取消任务** | `pool.terminate()`（暴力） | `future.cancel()`（优雅） |
| **超时** | `pool.map()` 不支持超时 | `future.result(timeout=30)` |
| **异常传播** | 子进程异常可能被吞 | `future.result()` 会抛出 |
| **Shutdown** | `close()` + `join()` 两步 | `with` 语句自动管理 |
| **内存** | 参数会 pickle 传给子进程 | 同样会 pickle |
| **何时用** | 老代码兼容、需要 `imap_unordered` | **新代码推荐**——更简洁、更安全 |

## 五、混合方案：asyncio + 多进程

```python
import asyncio
from concurrent.futures import ProcessPoolExecutor
import time


# ── 场景：Web 服务接收请求 → 异步接收 + 多进程处理 CPU 任务 ──

def cpu_heavy_task(data):
    """CPU 密集：图像预处理、特征提取等"""
    time.sleep(0.1)  # 模拟计算
    return {"result": sum(data)}


async def handle_request(executor: ProcessPoolExecutor, data):
    """异步接收请求，CPU 任务交给进程池"""
    loop = asyncio.get_running_loop()

    # 把 CPU 任务 offload 到进程池（不阻塞事件循环）
    result = await loop.run_in_executor(executor, cpu_heavy_task, data)
    return result


async def main():
    # 进程池在 async 世界外独立运行
    with ProcessPoolExecutor(max_workers=4) as executor:
        # 模拟 10 个并发请求
        tasks = [
            handle_request(executor, list(range(i * 1000, (i + 1) * 1000)))
            for i in range(10)
        ]
        results = await asyncio.gather(*tasks)
    return results


if __name__ == "__main__":
    asyncio.run(main())
```

## 六、常见陷阱与解决方案

### 陷阱一：pickle 序列化失败

```python
# ❌ lambda 不能 pickle
with Pool(4) as pool:
    pool.map(lambda x: x * 2, range(10))  # AttributeError: Can't pickle

# ✅ 用模块级函数
def double(x): return x * 2
with Pool(4) as pool:
    pool.map(double, range(10))


# ✅ ProcessPoolExecutor 也有同样限制
with ProcessPoolExecutor() as executor:
    # executor.submit(lambda x: x*2, 5)  # ❌ 同样失败
    executor.submit(double, 5)           # ✅
```

### 陷阱二：大对象频繁序列化

```python
# ❌ 每次 map 都把整个数据集 pickle 给每个子进程
huge_data = np.random.randn(100000, 1000)
with Pool(4) as pool:
    pool.map(process, huge_data)  # 4 个子进程 × 100000 次 pickle

# ✅ 用多进程共享内存（只拷贝指针，不拷贝数据）
def process_shared(shared_arr, index):
    return shared_arr[index].sum()

# 方法一：multiprocessing.shared_memory（Python 3.8+）
from multiprocessing import shared_memory

def process_shm(shm_name, shape, dtype, index):
    existing_shm = shared_memory.SharedMemory(name=shm_name)
    arr = np.ndarray(shape, dtype=dtype, buffer=existing_shm.buf)
    return arr[index].sum()

# 主进程
arr = np.random.randn(100000, 1000)
shm = shared_memory.SharedMemory(create=True, size=arr.nbytes)
shared_arr = np.ndarray(arr.shape, dtype=arr.dtype, buffer=shm.buf)
np.copyto(shared_arr, arr)

with Pool(4) as pool:
    indices = list(range(1000))
    pool.starmap(process_shm, [(shm.name, arr.shape, arr.dtype, i) for i in indices])
```

### 陷阱三：子进程不响应 Ctrl+C

```python
# ❌ pool.join() 会阻塞，Ctrl+C 无响应
pool = Pool(4)
pool.map(work, huge_list)  # 卡住
pool.close()
pool.join()

# ✅ 用 signal 处理
import signal

def init_worker():
    signal.signal(signal.SIGINT, signal.SIG_IGN)  # 子进程忽略 Ctrl+C

pool = Pool(4, initializer=init_worker)
try:
    pool.map(work, huge_list)
except KeyboardInterrupt:
    pool.terminate()
    pool.join()
```

### 陷阱四：线程池里的共享状态

```python
# ❌ 多线程写同一个变量需要加锁
counter = 0
def increment():
    global counter
    counter += 1  # 非原子操作！可能丢更新

with ThreadPoolExecutor(10) as executor:
    executor.map(lambda _: increment(), range(1000))
print(counter)  # 可能 < 1000


# ✅ 线程安全的方式
import threading
lock = threading.Lock()

def safe_increment():
    global counter
    with lock:
        counter += 1

# ✅ 或使用 concurrent.futures 的返回结果，在主线程聚合
with ThreadPoolExecutor(10) as executor:
    results = list(executor.map(lambda x: x * 2, range(1000)))
total = sum(results)  # 主线程做聚合，无竞争
```

